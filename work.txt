


c++20으로 플랫폼 독립적인 크로스 플랫폼 랜더러이고 첫번째로 vma와 vulkan hpp를 이용한 VulkanRenderer만 일단 다음과 같은 특징으로 객체지향으로 구현한다.
1. CreateXXX시 플랫폼 독립적인 XXXDesc매개변수를 사용하며 필요하거나 부족하면 플랫폼 독립적으로 확장.
2. Vulkan의 거의 모든 옵션을 XXXDesc에 맵핑한다.
3. CreateXXX, UpdateXXX, BindXXX 인터페이스 형식를 사용하고 
4. 모든 리소스는 핸들로 관리한다.
6 모든 리소스는 해지는  ReleaseResource하나만 사용하게 한다.
7. IRenderer내의 static Create함수로 VulakanRenderer를 인스턴스화.
8. 플랫폼 독립적인 IRenderer는 vulkan 관련 의존성이 없다.
9. 모든 코드는 Lumora 네임스페이스를 사용한다.
10. 라이브러리의 파일은 IRenderer.h, VulkanRenderer.h, VulkanRenderer.cpp 3개 파일이 전부다. 예시는 testXXX.cpp로 만들어줘

초기목표는 플랫폼 독립적인 IRenderer만 사용하여 유니폼버퍼를 사용하고 텍스쳐링된 인덱스된 삼각형을 그리는 테스트 코드를 목표로한다. 
"Lumora": 빛(Lumen)과 오라(Aura)의 조합으로, 렌더링의 시각적 아름다움과 혁신을 상징합니다.

    Usage usage;
};

struct PipelineDesc {
    ShaderHandle vertexShader;       // 버텍스 셰이더
    ShaderHandle fragmentShader;     // 프래그먼트 셰이더
    ShaderHandle computeShader;      // 컴퓨트 셰이더 (옵션)
    VertexLayoutHandle vertexLayout; // 플랫폼 독립적 버텍스 레이아웃
   
    RenderPassHandle renderPassHandle;

    struct DescriptorBindingDesc {
        uint32_t binding; // UniformBuffer, 
        DescriptorType descriptorType; 
        ShaderStageFlags stageFlags; 
        uint32_t descriptorCount = 1; 
    };
    struct DescriptorSetLayoutDesc {         std::vector<DescriptorBindingDesc> bindings;
    }; 
struct PushConstantRangeDesc { 
    ShaderStageFlags stageFlags; 
    uint32_t offset; uint32_t size;
}; 
    std::vector<DescriptorSetLayoutDesc> descriptorSetLayouts;     std::vector<PushConstantRangeDesc> pushConstantRanges;

    struct ViewportDesc {
        float x = 0.0f;
        float y = 0.0f;
        float width = 0.0f;
        float height = 0.0f;
        float minDepth = 0.0f;
        float maxDepth = 1.0f;
    } viewport;

    struct ScissorDesc {
    int32_t offsetX = 0;
    int32_t offsetY = 0;
    uint32_t width = 0;
    uint32_t height = 0;
    } scissor;

    struct RasterizationState {
        bool depthClampEnable = false;
        bool rasterizerDiscardEnable = false;
        PolygonMode polygonMode = PolygonMode::Fill;
        CullMode cullMode = CullMode::Back;
        FrontFace frontFace = FrontFace::Clockwise;
    } rasterizationState;

    struct ColorBlendState {
        bool blendEnable = false;
        BlendFactor srcColorBlendFactor = BlendFactor::One;
        BlendFactor dstColorBlendFactor = BlendFactor::Zero;
        BlendOp colorBlendOp = BlendOp::Add;
        BlendFactor srcAlphaBlendFactor = BlendFactor::One;
        BlendFactor dstAlphaBlendFactor = BlendFactor::Zero;
        BlendOp alphaBlendOp = BlendOp::Add;
    };

    std::vector<ColorBlendState> colorBlendStates; // 각 출력 컬러에 대한 블렌드 상태

    struct DepthStencilState {
        bool depthTestEnable = true;
        bool depthWriteEnable = true;
        CompareOp depthCompareOp = CompareOp::Less;
        bool stencilTestEnable = false;
    } depthStencilState;

    struct MultisampleState {
        int sampleCount = 1; // 멀티샘플링 지원
    } multisampleState;

    enum class PrimitiveTopology {
        TriangleList, TriangleStrip, LineList
    } topology = PrimitiveTopology::TriangleList;
};
